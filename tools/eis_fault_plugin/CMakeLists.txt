cmake_minimum_required(VERSION 3.25)
project(issue12_eis_fault VERSION 1.0 LANGUAGES CXX)

set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(KWin 6.7.5 EXACT REQUIRED CONFIG)
add_library(issue12_eis_fault MODULE main.cpp)
set_target_properties(issue12_eis_fault PROPERTIES PREFIX "")
target_link_libraries(issue12_eis_fault PRIVATE KWin::kwin Qt6::DBus)
